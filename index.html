// ========================================== // 1. ИНИЦИАЛИЗАЦИЯ И НАСТРОЙКИ POCKETBASE // ========================================== // Укажите хост вашей локальной или удаленной базы PocketBase const POCKETBASE_URL = 'http://127.0.0.1:8090'; const pb = new PocketBase(POCKETBASE_URL); // Отключаем авто-отмену запросов для стабильности pb.autoCancellation(false); // ========================================== // 2. УПРАВЛЕНИЕ РАДИОПЛЕЕРОМ // ========================================== const audioPlayer = document.getElementById('audioPlayer'); const statusText = document.getElementById('statusText'); /** * Запуск воспроизведения выбранного потока * @param {string} url - Прямая ссылка на аудиопоток * @param {string} stationName - Название станции для статуса */ function playStream(url, stationName) { if (!audioPlayer) return; audioPlayer.src = url; audioPlayer.load(); const playPromise = audioPlayer.play(); if (playPromise !== undefined) { playPromise .then(() => { if (statusText) statusText.textContent = `Сейчас играет: ${stationName}`; }) .catch(error => { console.warn("Ошибка воспроизведения аудио:", error); if (statusText) statusText.textContent = `Пауза: Нажмите Play для ${stationName}`; }); } } // ========================================== // 3. АВТОРИЗАЦИЯ ОAUTH2 (GOOGLE) // ========================================== async function loginGoogle() { try { const authData = await pb.collection('users').authWithOAuth2({ provider: 'google' }); // Обновляем аватарку и имя в шапке/профиле при успешном входе const user = authData.record; alert(`Добро пожаловать, ${user.name || user.username || user.email}!`); // Обновляем UI после входа updateUserProfileUI(user); } catch (err) { console.error("Ошибка авторизации через Google:", err); alert("Не удалось войти через Google. Проверьте настройки OAuth2 в панели PocketBase."); } } function updateUserProfileUI(user) { const btnGoogle = document.querySelector('.btn-google'); if (btnGoogle && user) { btnGoogle.textContent = user.name || 'Профиль'; btnGoogle.onclick = openProfileModal; } } // ========================================== // 4. ЛОГИКА ЧАТА И REALTIME // ========================================== const msgInput = document.getElementById('msgInput'); const chatMessages = document.getElementById('chatMessages'); /** * Отправка сообщения в PocketBase или локально */ async function sendMessage() { if (!msgInput) return; const text = msgInput.value.trim(); if (!text) return; // Очищаем поле ввода msgInput.value = ''; try { // Попытка отправить в PocketBase await pb.collection('messages').create({ text: text, user: pb.authStore.model ? pb.authStore.model.id : null }); } catch (err) { console.warn("PocketBase недоступен или произошла ошибка. Сообщение отправлено в локальный режим:", err); // Локальный фолбэк для демонстрации const currentAuthor = pb.authStore.model ? (pb.authStore.model.name || pb.authStore.model.username) : 'Гость'; appendMessage(currentAuthor, text); } } /** * Отрисовка нового сообщения в окне чата */ function appendMessage(author, text) { if (!chatMessages) return; const msgDiv = document.createElement('div'); msgDiv.className = 'chat-msg'; // Экранирование HTML от XSS const safeAuthor = escapeHtml(author); const safeText = escapeHtml(text); msgDiv.innerHTML = `
${safeAuthor}
${safeText}`; chatMessages.appendChild(msgDiv); // Автоскролл вниз к последнему сообщению chatMessages.scrollTop = chatMessages.scrollHeight; } /** * Обработка нажатия Enter в поле ввода */ function handleKeyPress(event) { if (event.key === 'Enter') { sendMessage(); } } /** * Вспомогательная функция защиты от XSS */ function escapeHtml(str) { return String(str) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } // ========================================== // 5. ИНИЦИАЛИЗАЦИЯ И REALTIME ПОДПИСКА // ========================================== document.addEventListener('DOMContentLoaded', () => { // Проверка текущей сессии if (pb.authStore.isValid && pb.authStore.model) { updateUserProfileUI(pb.authStore.model); } // Подписка на коллекцию messages в режиме реального времени try { pb.collection('messages').subscribe('*', function (e) { if (e.action === 'create') { const authorName = e.record.expand && e.record.expand.user ? (e.record.expand.user.name || e.record.expand.user.username) : 'Участник'; appendMessage(authorName, e.record.text); } }); } catch (e) { console.log("Realtime подписка ожидает подключения к PocketBase."); } });
